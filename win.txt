function q { exit }

Set-PSReadLineKeyHandler -Key Ctrl+f -ScriptBlock {
    $dirs = @(
        "$HOME",
        "$HOME\dev",
        "$HOME\.config"
    )

    $selected = $dirs | ForEach-Object {
        Get-ChildItem $_ -Directory -Recurse -Depth 2 -ErrorAction SilentlyContinue
    } | Select-Object -ExpandProperty FullName | fzf

    if ($selected) {
        # Insere o comando 'cd "caminho"; Clear-Host' no buffer
        [Microsoft.PowerShell.PSConsoleReadLine]::Insert("cd `"$selected`"; Clear-Host")
        [Microsoft.PowerShell.PSConsoleReadLine]::AcceptLine()
    }
}

}

function gg {
    param(
        [Parameter(ValueFromRemainingArguments = $true)]
        [string[]]$Termo
    )

    if (-not $Termo) {
        Write-Host "Uso: gg termo de pesquisa"
        return
    }

    $query = ($Termo -join " ")
    $url = "https://www.google.com/search?q=" + [uri]::EscapeDataString($query)
    Start-Process $url
}

function yt {
    param([Parameter(ValueFromRemainingArguments = $true)][string[]]$Termo)
    if (-not $Termo) { Write-Host "Uso: yt termo de pesquisa"; return }
    $query = ($Termo -join " ")
    Start-Process ("https://www.youtube.com/results?search_query=" + [uri]::EscapeDataString($query))
}

function gh {
    param([Parameter(ValueFromRemainingArguments = $true)][string[]]$Termo)
    if (-not $Termo) { Write-Host "Uso: gh termo de pesquisa"; return }
    $query = ($Termo -join " ")
    Start-Process ("https://github.com/search?q=" + [uri]::EscapeDataString($query) + "&type=repositories")
}

function so {

    Start-Process ("https://stackoverflow.com/search?q=" + [uri]::EscapeDataString($query))
}
